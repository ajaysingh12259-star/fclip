    96→const YT_ID_RE = /(?:youtube\.com\/(?:watch\?(?:[^#]*&)?v=|shorts\/|embed\/|live\/)|youtu\.be\/)([\w-]{11})/;
    97→
    98→function parseVideoId(url: string): string | null {
    99→  const m = url.match(YT_ID_RE);
   100→  return m ? m[1] : null;
   101→}
   102→
   103→function extractT(url: string): number | null {
   104→  const m = url.match(/[?&](?:t|start)=([\dhms.]+)/);
   105→  if (!m) return null;
   106→  const raw = m[1];
   107→  if (/^\d+(\.\d+)?$/.test(raw)) return parseFloat(raw);
   108→  const h = raw.match(/(\d+(?:\.\d+)?)h/);
   109→  const min = raw.match(/(\d+(?:\.\d+)?)m/);
   110→  const s = raw.match(/(\d+(?:\.\d+)?)s/);
   111→  const v = (h ? parseFloat(h[1]) * 3600 : 0) + (min ? parseFloat(min[1]) * 60 : 0) + (s ? parseFloat(s[1]) : 0);
   112→  return v > 0 ? v : null;
   113→}
   114→
   115→function parseTimeInput(text: string): number | null {
   116→  const t = text.trim();
   117→  if (!t) return null;
   118→  if (/^\d+(\.\d+)?$/.test(t)) return parseFloat(t);
   119→  const parts = t.split(":").map((p) => parseFloat(p));
   120→  if (parts.some((p) => isNaN(p) || p < 0)) return null;
   121→  let sec = 0;
   122→  for (const p of parts) sec = sec * 60 + p;
   123→  return sec;
   124→}
   125→
   126→function fmtClock(t: number): string {
   127→  const s = Math.floor(t % 60);
   128→  const m = Math.floor((t / 60) % 60);
   129→  const h = Math.floor(t / 3600);
   130→  const mm = h > 0 ? String(m).padStart(2, "0") : String(m);
   131→  return (h > 0 ? `${h}:` : "") + `${mm}:${String(s).padStart(2, "0")}`;
   132→}
   133→
   134→function fmtBytes(b: number | null): string {
   135→  if (!b || b <= 0) return "—";
   136→  if (b < 1024 * 1024) return `${Math.round(b / 1024)} KB`;
   137→  if (b < 1024 * 1024 * 1024) return `${(b / (1024 * 1024)).toFixed(1)} MB`;
   138→  return `${(b / (1024 * 1024 * 1024)).toFixed(2)} GB`;
   139→}
   140→
   141→const BITRATE_TABLE: Record<number, number> = { 2160: 22, 1440: 11, 1080: 6, 720: 3, 480: 1.5 };
   142→
   143→function estimateSizeMB(
   144→  start: number,
   145→  end: number,
   146→  quality: string,
   147→  qualities: { height: number; tbr: number | null }[]
   148→): number {
   149→  const dur = Math.max(0, end - start);
   150→  let mbps = 8;
   151→  if (quality === "best") {
   152→    const top = qualities[0];
   153→    if (top) mbps = top.tbr ? top.tbr / 1000 : BITRATE_TABLE[top.height] ?? 8;
   154→  } else {
   155→    const h = Number(quality);
   156→    const q = qualities.find((x) => x.height === h);
   157→    mbps = q ? (q.tbr ? q.tbr / 1000 : BITRATE_TABLE[h] ?? 5) : BITRATE_TABLE[h] ?? 5;
   158→  }
   159→  return Math.max(1, Math.round((dur * mbps) / 8));
   160→}

   162→async function jsonFetch<T>(url: string, init?: RequestInit): Promise<{ ok: boolean; data: T }> {
   163→  const res = await fetch(url, init);
   164→  const data = (await res.json()) as T;
   165→  return { ok: res.ok, data };
   166→}
